    const COLS = 10, ROWS = 20;
let board = grid(ROWS, COLS);
let piece = spawn(), score = 0;

function spawn() {
  const k = "IJLOSTZ"[rand(7)];
  return { k, x: 3, y: 0, r: 0 };
}

function fits(p) {
  return cells(p).every(([x, y]) =>
    x >= 0 && x < COLS && y < ROWS && !board[y]?.[x]);
}

function clearLines() {
  for (let y = ROWS - 1; y >= 0; y--) {
    if (board[y].every(Boolean)) {
      board.splice(y, 1);
      board.unshift(Array(COLS).fill(0));
      score += 100; y++;
    }
  }
}

function loop(t) {
  if (t - last > speed) drop();
  draw(board, piece);
  requestAnimationFrame(loop);
}
